cmake_minimum_required(VERSION 3.20)
project(PSICConsumer LANGUAGES C CXX)
if(PSIC_SOURCE_DIR)
    set(PSIC_BUILD_CLI OFF CACHE BOOL "" FORCE)
    add_subdirectory("${PSIC_SOURCE_DIR}" psic)
    if(TARGET psic_library_tests)
        message(FATAL_ERROR "Embedding PSIC unexpectedly enabled its tests")
    endif()
else()
    find_package(PSIC 0.1 CONFIG REQUIRED)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE PSIC::psic)
enable_testing()
add_test(NAME consumer COMMAND consumer)
